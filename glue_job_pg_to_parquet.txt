import sys
import logging
import time
import psycopg2
import pandas as pd
from datetime import datetime
from awsglue.utils import getResolvedOptions
from awsglue.context import GlueContext
from pyspark.context import SparkContext

# Accept the arguments passed to the Glue job
args = getResolvedOptions(sys.argv, [
    'CONFIG_DB_HOST', 'CONFIG_DB_PORT', 'CONFIG_DB_NAME', 'CONFIG_DB_USER', 'CONFIG_DB_PASSWORD',
    'WORKFLOW_DB_HOST', 'WORKFLOW_DB_PORT', 'WORKFLOW_DB_NAME', 'WORKFLOW_DB_USER', 'WORKFLOW_DB_PASSWORD',
    'QUERY_TABLE', 'OUTPUT_PATH'
])

# Set up logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Initialize Spark & Glue contexts
sc = SparkContext()
glueContext = GlueContext(sc)

# === Fetching Configuration Query from Config DB ===
def get_pg_connection(host, port, db_name, user, password):
    try:
        conn = psycopg2.connect(
            host=host,
            port=port,
            dbname=db_name,
            user=user,
            password=password
        )
        return conn
    except Exception as e:
        logger.error(f"Error connecting to database: {e}")
        raise

def fetch_query_from_config_db(connection, query_table):
    query = f"SELECT query FROM {query_table} LIMIT 1;"
    df = pd.read_sql(query, connection)
    if not df.empty:
        return df.iloc[0]['query']
    else:
        logger.error("No query found in config table.")
        raise ValueError("No query found in config table.")

def fetch_data_from_workflow_db(query, connection):
    return pd.read_sql(query, connection)

        os.makedirs(output_path)

    full_path = os.path.join(output_path, filename)
    df.to_parquet(full_path, index=False)
    logger.info(f"Saved: {full_path}")

def main():
    try:
        # Connect to Config DB to fetch query
        config_conn = get_pg_connection(
            args['CONFIG_DB_HOST'], args['CONFIG_DB_PORT'], args['CONFIG_DB_NAME'],
            args['CONFIG_DB_USER'], args['CONFIG_DB_PASSWORD']
        )
        logger.info("Connected to Config database.")

        # Fetch the query to be executed in workflow DB
        query = fetch_query_from_config_db(config_conn, args['QUERY_TABLE'])
        logger.info(f"Fetched query from Config DB: {query}")

        # Connect to Workflow DB and execute the fetched query
        workflow_conn = get_pg_connection(
            args['WORKFLOW_DB_HOST'], args['WORKFLOW_DB_PORT'], args['WORKFLOW_DB_NAME'],
            args['WORKFLOW_DB_USER'], args['WORKFLOW_DB_PASSWORD']
        )
        logger.info("Connected to Workflow database.")

        # Fetch data from Workflow DB using the query
        df = fetch_data_from_workflow_db(query, workflow_conn)

        # Generate output filename
        output_filename = f"workflow_data_{datetime.now().strftime('%Y%m%d_%H%M%S')}.parquet"
        save_parquet(df, output_filename, args['OUTPUT_PATH'])

        # Commit the Glue job to indicate it is complete
        glueContext.commit()

    except Exception as e:
        logger.error(f"Error: {e}")
        # Commit is not called in case of error to ensure the job fails
    finally:
        if 'config_conn' in locals():
            config_conn.close()
            logger.info("Config database connection closed.")
        if 'workflow_conn' in locals():
            workflow_conn.close()
            logger.info("Workflow database connection closed.")

if __name__ == "__main__":
    main()
